cmake_minimum_required(VERSION 3.25)
project(lrclexec VERSION 0.2.0 LANGUAGES CXX)

include(GNUInstallDirs)
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()

set(LRCLEXEC_LEXEC_SOURCE_DIR "" CACHE PATH "Local lexec source tree")
set(LRCLEXEC_CO2_SOURCE_DIR "" CACHE PATH "Local co2 source tree for the Nav2 example")
set(build_tests_default OFF)
if(PROJECT_IS_TOP_LEVEL AND BUILD_TESTING)
    set(build_tests_default ON)
endif()
option(LRCLEXEC_BUILD_TESTS "Build ROS integration tests" ${build_tests_default})
option(LRCLEXEC_BUILD_EXAMPLES "Build examples" ${PROJECT_IS_TOP_LEVEL})
option(LRCLEXEC_BUILD_NAV2_EXAMPLE "Build the optional Nav2 message example and tests" OFF)
option(LRCLEXEC_BUILD_MUJOCO_SIM "Build the headless MuJoCo/Nav2 simulation" OFF)
option(LRCLEXEC_INSTALL "Install the ROS/CMake package" ${PROJECT_IS_TOP_LEVEL})
option(LRCLEXEC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(bundle_lexec_default OFF)
if(LRCLEXEC_LEXEC_SOURCE_DIR)
    set(bundle_lexec_default ON)
endif()
option(LRCLEXEC_BUNDLE_LEXEC_HEADERS "Install the provided lexec core headers" ${bundle_lexec_default})

if(NOT TARGET lexec::lexec)
    if(LRCLEXEC_LEXEC_SOURCE_DIR)
        set(LEXEC_BUILD_TESTS OFF CACHE BOOL "Build lexec tests")
        add_subdirectory("${LRCLEXEC_LEXEC_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lexec" EXCLUDE_FROM_ALL)
    else()
        find_package(lexec CONFIG REQUIRED)
    endif()
endif()
if(LRCLEXEC_BUNDLE_LEXEC_HEADERS AND NOT EXISTS "${LRCLEXEC_LEXEC_SOURCE_DIR}/include/lexec/execution.hpp")
    message(FATAL_ERROR "Bundling requires LRCLEXEC_LEXEC_SOURCE_DIR with the matching lexec headers")
endif()
if(LRCLEXEC_INSTALL)
    find_package(ament_cmake REQUIRED)
    if(NOT LRCLEXEC_BUNDLE_LEXEC_HEADERS)
        find_package(lexec CONFIG REQUIRED)
    endif()
endif()

find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(Threads REQUIRED)
add_subdirectory(lrclexec)

if(LRCLEXEC_BUILD_TESTS OR LRCLEXEC_BUILD_EXAMPLES)
    find_package(example_interfaces REQUIRED)
endif()
if(LRCLEXEC_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
if(LRCLEXEC_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
if(LRCLEXEC_BUILD_NAV2_EXAMPLE)
    find_package(nav2_msgs REQUIRED)
    if(NOT TARGET co2::co2)
        if(LRCLEXEC_CO2_SOURCE_DIR)
            set(CO2_BUILD_TESTS OFF CACHE BOOL "Build co2 tests")
            add_subdirectory("${LRCLEXEC_CO2_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/co2" EXCLUDE_FROM_ALL)
        else()
            find_package(co2 CONFIG REQUIRED)
        endif()
    endif()
    add_subdirectory(examples/nav2)
endif()
if(LRCLEXEC_BUILD_MUJOCO_SIM)
    if(NOT LRCLEXEC_BUILD_NAV2_EXAMPLE)
        message(FATAL_ERROR "MuJoCo simulation requires LRCLEXEC_BUILD_NAV2_EXAMPLE=ON")
    endif()
    add_subdirectory(simulation/mujoco)
endif()
if(LRCLEXEC_INSTALL)
    ament_export_targets(lrclexecTargets HAS_LIBRARY_TARGET NAMESPACE lrclexec::)
    ament_export_dependencies(rclcpp rclcpp_action Threads)
    ament_package(CONFIG_EXTRAS cmake/lrclexec-dependencies.cmake.in
                  CONFIG_EXTRAS_POST cmake/lrclexec-lexec.cmake.in)
endif()
