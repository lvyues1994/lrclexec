find_package(rclcpp_lifecycle REQUIRED)
add_executable(lifecycle_example main.cpp)
target_link_libraries(lifecycle_example PRIVATE lrclexec::lrclexec rclcpp_lifecycle::rclcpp_lifecycle)
target_compile_options(lifecycle_example PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror=return-type)
if(LRCLEXEC_INSTALL AND LRCLEXEC_BUILD_EXAMPLES)
    install(TARGETS lifecycle_example RUNTIME DESTINATION lib/${PROJECT_NAME})
endif()
if(LRCLEXEC_BUILD_TESTS)
    set(session_modes single multi)
    if(LRCLEXEC_TEST_EVENTS_LIFECYCLE)
        list(APPEND session_modes events)
    endif()
    foreach(mode IN LISTS session_modes)
        add_test(NAME lifecycle_session_${mode} COMMAND lifecycle_example --executor ${mode} --cycles 3)
        set_tests_properties(lifecycle_session_${mode} PROPERTIES TIMEOUT 30
            ENVIRONMENT "ROS_DOMAIN_ID=231;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST;ROS_LOG_DIR=/tmp/lrclexec-session-logs"
            RESOURCE_LOCK lrclexec_session_dds)
    endforeach()
    if(LRCLEXEC_TEST_EVENTS_LIFECYCLE)
        set_tests_properties(lifecycle_session_events PROPERTIES LABELS "experimental;sdk-limitation")
    endif()
endif()
